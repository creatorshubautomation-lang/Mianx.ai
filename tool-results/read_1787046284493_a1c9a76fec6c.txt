   172→function formatDate(dateStr: string): string {
   173→  return new Date(dateStr).toLocaleDateString('en-US', {
   174→    month: 'short',
   175→    day: 'numeric',
   176→    year: 'numeric',
   177→  })
   178→}
   179→
   180→function formatDateTime(dateStr: string | null): string {
   181→  if (!dateStr) return '—'
   182→  return new Date(dateStr).toLocaleString('en-US', {
   183→    month: 'short',
   184→    day: 'numeric',
   185→    hour: '2-digit',
   186→    minute: '2-digit',
   187→  })
   188→}

   190→function getStepCount(definition: string): number {
   191→  const parsed = parseJsonField<Record<string, unknown>>(definition, {})
   192→  const steps = parsed?.steps as unknown[] | undefined
   193→  if (Array.isArray(steps)) return steps.length
   194→  return 0
   195→}
   196→
   197→function formatDuration(startedAt: string | null, completedAt: string | null): string {
   198→  if (!startedAt) return '—'
   199→  const start = new Date(startedAt).getTime()
   200→  const end = completedAt ? new Date(completedAt).getTime() : Date.now()
   201→  const diffMs = end - start
   202→  const diffSec = Math.floor(diffMs / 1000)
   203→  if (diffSec < 60) return `${diffSec}s`
   204→  const diffMin = Math.floor(diffSec / 60)
   205→  const remainingSec = diffSec % 60
   206→  if (diffMin < 60) return `${diffMin}m ${remainingSec}s`
   207→  const diffHr = Math.floor(diffMin / 60)
   208→  return `${diffHr}h ${diffMin % 60}m`
   209→}

   233→function CreateWorkflowDialog({
   234→  open,
   235→  onOpenChange,
   236→  onCreated,
   237→  orgId,
   238→}: {
   239→  open: boolean
   240→  onOpenChange: (open: boolean) => void
   241→  onCreated: () => void
   242→  orgId: string
   243→}) {
   244→  const [name, setName] = useState('')
   245→  const [slug, setSlug] = useState('')
   246→  const [triggerType, setTriggerType] = useState('manual')
   247→  const [definition, setDefinition] = useState('')
   248→  const [submitting, setSubmitting] = useState(false)
   249→
   250→  const handleNameChange = (val: string) => {
   251→    setName(val)
   252→    if (!slug || slug === slugify(name)) {
   253→      setSlug(slugify(val))
   254→    }
   255→  }
   256→
   257→  const handleSubmit = async () => {
   258→    if (!name.trim() || !slug.trim()) return
   259→    setSubmitting(true)
   260→    try {
   261→      let parsedDef: Record<string, unknown> = {}
   262→      if (definition.trim()) {
   263→        try {
   264→          parsedDef = JSON.parse(definition)
   265→        } catch {
   266→          toast({
   267→            title: 'Invalid JSON',
   268→            description: 'Please check your workflow definition JSON.',
   269→            variant: 'destructive',
   270→          })
   271→          setSubmitting(false)
   272→          return
   273→        }
   274→      }
   275→
   276→      const res = await fetch(`/api/workflows?organizationId=${orgId}`, {
   277→        method: 'POST',
   278→        headers: { 'Content-Type': 'application/json' },
   279→        body: JSON.stringify({
   280→          name: name.trim(),
   281→          slug: slug.trim(),
   282→          triggerType,
   283→          definition: Object.keys(parsedDef).length > 0 ? parsedDef : undefined,
   284→        }),
   285→      })
   286→      if (res.ok) {
   287→        toast({ title: 'Workflow created', description: `${name} has been created.` })
   288→        setName('')
   289→        setSlug('')
   290→        setTriggerType('manual')
   291→        setDefinition('')
   292→        onOpenChange(false)
   293→        onCreated()
   294→      } else {
   295→        const json = await res.json().catch(() => ({}))
   296→        toast({
   297→          title: 'Failed to create workflow',
   298→          description: json?.error?.message ?? 'An error occurred.',
   299→          variant: 'destructive',
   300→        })
   301→      }
   302→    } catch {
   303→      toast({ title: 'Error', description: 'Network error.', variant: 'destructive' })
   304→    } finally {
   305→      setSubmitting(false)
   306→    }
   307→  }
   308→
   309→  return (
   310→    <Dialog open={open} onOpenChange={onOpenChange}>
   311→      <DialogContent className="glass-strong border-[rgba(99,102,241,0.18)] max-w-lg">
   312→        <DialogHeader>
   313→          <DialogTitle className="text-[#e2e8f0]">Create Workflow</DialogTitle>
   314→        </DialogHeader>
   315→        <div className="space-y-4 mt-2">
   316→          <div className="space-y-2">
   317→            <Label className="text-xs text-[#94a3b8]">Name</Label>
   318→            <Input
   319→              value={name}
   320→              onChange={(e) => handleNameChange(e.target.value)}
   321→              placeholder="e.g. Daily Report Generator"
   322→              className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#64748b]"
   323→            />
   324→          </div>
   325→          <div className="space-y-2">
   326→            <Label className="text-xs text-[#94a3b8]">Slug</Label>
   327→            <Input
   328→              value={slug}
   329→              onChange={(e) => setSlug(e.target.value)}
   330→              placeholder="daily-report-generator"
   331→              className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#64748b]"
   332→            />
   333→          </div>
   334→          <div className="space-y-2">
   335→            <Label className="text-xs text-[#94a3b8]">Trigger Type</Label>
   336→            <Select value={triggerType} onValueChange={setTriggerType}>
   337→              <SelectTrigger className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0]">
   338→                <SelectValue />
   339→              </SelectTrigger>
   340→              <SelectContent className="glass-strong border-[rgba(99,102,241,0.18)]">
   341→                {TRIGGER_TYPES.map((t) => {
   342→                  const cfg = TRIGGER_CONFIG[t]
   343→                  return (
   344→                    <SelectItem
   345→                      key={t}
   346→                      value={t}
   347→                      className="text-[#e2e8f0] focus:bg-[rgba(99,102,241,0.12)] focus:text-white"
   348→                    >
   349→                      {cfg.label}
   350→                    </SelectItem>
   351→                  )
   352→                })}
   353→              </SelectContent>
   354→            </Select>
   355→          </div>
   356→          <div className="space-y-2">
   357→            <Label className="text-xs text-[#94a3b8]">Definition (JSON)</Label>
   358→            <Textarea
   359→              value={definition}
   360→              onChange={(e) => setDefinition(e.target.value)}
   361→              placeholder={WORKFLOW_DEFINITION_PLACEHOLDER}
   362→              rows={8}
   363→              className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#64748b] font-mono text-xs resize-none"
   364→            />
   365→          </div>
   366→          <div className="flex justify-end gap-3 pt-2">
   367→            <Button
   368→              variant="outline"
   369→              className="border-[rgba(99,102,241,0.15)] text-[#94a3b8] hover:text-[#e2e8f0]"
   370→              onClick={() => onOpenChange(false)}
   371→            >
   372→              Cancel
   373→            </Button>
   374→            <Button
   375→              className="btn-gradient text-white"
   376→              onClick={handleSubmit}
   377→              disabled={!name.trim() || !slug.trim() || submitting}
   378→            >
   379→              {submitting ? 'Creating...' : 'Create Workflow'}
   380→            </Button>
   381→          </div>
   382→        </div>
   383→      </DialogContent>
   384→    </Dialog>
   385→  )
   386→}

   472→export default function WorkflowsView() {
   473→  const activeOrg = useActiveOrg()
   474→  const isAuthenticated = useStore((s) => s.isAuthenticated)
   475→
   476→  const [workflows, setWorkflows] = useState<WorkflowDto[]>([])
   477→  const [loading, setLoading] = useState(true)
   478→  const [error, setError] = useState<string | null>(null)
   479→  const [createOpen, setCreateOpen] = useState(false)
   480→  const [runsWorkflow, setRunsWorkflow] = useState<WorkflowDto | null>(null)
   481→  const [runsOpen, setRunsOpen] = useState(false)
   482→  const [runsData, setRunsData] = useState<WorkflowRunDto[]>([])
   483→  const [runsLoading, setRunsLoading] = useState(false)
   484→  const [runningIds, setRunningIds] = useState<Set<string>>(new Set())
   485→
   486→  const fetchWorkflows = useCallback(async () => {
   487→    if (!activeOrg?.id) return
   488→    setLoading(true)
   489→    setError(null)
   490→    try {
   491→      const res = await fetch(
   492→        `/api/workflows?organizationId=${activeOrg.id}&limit=50`
   493→      )
   494→      if (res.ok) {
   495→        const json = await res.json()
   496→        setWorkflows(json.data ?? [])
   497→      } else {
   498→        setError('Failed to load workflows')
   499→      }
   500→    } catch {
   501→      setError('Network error')
   502→    } finally {
   503→      setLoading(false)
   504→    }
   505→  }, [activeOrg?.id])
   506→
   507→  useEffect(() => {
   508→    if (activeOrg?.id) fetchWorkflows()
   509→  }, [activeOrg?.id, fetchWorkflows])
   510→
   511→  const handleRun = async (wf: WorkflowDto) => {
   512→    setRunningIds((prev) => new Set(prev).add(wf.id))
   513→    try {
   514→      const res = await fetch(`/api/workflows/${wf.id}/runs`, {
   515→        method: 'POST',
   516→        headers: { 'Content-Type': 'application/json' },
   517→        body: JSON.stringify({}),
   518→      })
   519→      if (res.ok) {
   520→        toast({
   521→          title: 'Workflow triggered',
   522→          description: `${wf.name} has been queued for execution.`,
   523→        })
   524→      } else {
   525→        toast({
   526→          title: 'Failed to trigger workflow',
   527→          variant: 'destructive',
   528→        })
   529→      }
   530→    } catch {
   531→      toast({ title: 'Error', description: 'Network error.', variant: 'destructive' })
   532→    } finally {
   533→      setRunningIds((prev) => {
   534→        const next = new Set(prev)
   535→        next.delete(wf.id)
   536→        return next
   537→      })
   538→    }
   539→  }
   540→
   541→  const handleViewRuns = async (wf: WorkflowDto) => {
   542→    setRunsWorkflow(wf)
   543→    setRunsData([])
   544→    setRunsLoading(true)
   545→    setRunsOpen(true)
   546→    try {
   547→      const res = await fetch(`/api/workflows/${wf.id}/runs?limit=20`)
   548→      if (res.ok) {
   549→        const json = await res.json()
   550→        setRunsData(json.data ?? [])
   551→      }
   552→    } catch {
   553→      setRunsData([])
   554→    } finally {
   555→      setRunsLoading(false)
   556→    }
   557→  }
   558→
   559→  const handleDelete = async (wf: WorkflowDto) => {
   560→    try {
   561→      const res = await fetch(
   562→        `/api/workflows/${wf.id}?organizationId=${activeOrg!.id}`,
   563→        { method: 'DELETE' }
   564→      )
   565→      if (res.ok) {
   566→        toast({ title: 'Workflow deleted', description: `${wf.name} has been removed.` })
   567→        fetchWorkflows()
   568→      } else {
   569→        toast({ title: 'Delete failed', variant: 'destructive' })
   570→      }
   571→    } catch {
   572→      toast({ title: 'Error', description: 'Network error.', variant: 'destructive' })
   573→    }
   574→  }
   575→
   576→  // Auth guard
   577→  if (!isAuthenticated) {
