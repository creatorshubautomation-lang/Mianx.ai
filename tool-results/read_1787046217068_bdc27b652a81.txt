   337→function CreateAgentDialog({
   338→  open,
   339→  onOpenChange,
   340→  onCreated,
   341→  orgId,
   342→}: {
   343→  open: boolean
   344→  onOpenChange: (open: boolean) => void
   345→  onCreated: () => void
   346→  orgId: string
   347→}) {
   348→  const [name, setName] = useState('')
   349→  const [slug, setSlug] = useState('')
   350→  const [description, setDescription] = useState('')
   351→  const [type, setType] = useState('assistant')
   352→  const [status, setStatus] = useState<AgentStatus>('draft')
   353→  const [capabilities, setCapabilities] = useState('')
   354→  const [submitting, setSubmitting] = useState(false)
   355→
   356→  const handleNameChange = (val: string) => {
   357→    setName(val)
   358→    if (!slug || slug === slugify(name)) {
   359→      setSlug(slugify(val))
   360→    }
   361→  }
   362→
   363→  const handleSubmit = async () => {
   364→    if (!name.trim() || !slug.trim()) return
   365→    setSubmitting(true)
   366→    try {
   367→      const caps = capabilities
   368→        .split(',')
   369→        .map((c) => c.trim())
   370→        .filter(Boolean)
   371→
   372→      const res = await fetch(`/api/agents?organizationId=${orgId}`, {
   373→        method: 'POST',
   374→        headers: { 'Content-Type': 'application/json' },
   375→        body: JSON.stringify({
   376→          name: name.trim(),
   377→          slug: slug.trim(),
   378→          description: description.trim() || undefined,
   379→          type,
   380→          status,
   381→          capabilities: caps.length > 0 ? caps : undefined,
   382→        }),
   383→      })
   384→      if (res.ok) {
   385→        toast({ title: 'Agent created', description: `${name} has been created successfully.` })
   386→        setName('')
   387→        setSlug('')
   388→        setDescription('')
   389→        setType('assistant')
   390→        setStatus('draft')
   391→        setCapabilities('')
   392→        onOpenChange(false)
   393→        onCreated()
   394→      } else {
   395→        const json = await res.json().catch(() => ({}))
   396→        toast({
   397→          title: 'Failed to create agent',
   398→          description: json?.error?.message ?? 'An unexpected error occurred.',
   399→          variant: 'destructive',
   400→        })
   401→      }
   402→    } catch {
   403→      toast({ title: 'Error', description: 'Network error. Please try again.', variant: 'destructive' })
   404→    } finally {
   405→      setSubmitting(false)
   406→    }
   407→  }
   408→
   409→  return (
   410→    <Dialog open={open} onOpenChange={onOpenChange}>
   411→      <DialogContent className="glass-strong border-[rgba(99,102,241,0.18)] max-w-lg">
   412→        <DialogHeader>
   413→          <DialogTitle className="text-[#e2e8f0]">Create Agent</DialogTitle>
   414→        </DialogHeader>
   415→        <div className="space-y-4 mt-2">
   416→          <div className="space-y-2">
   417→            <Label className="text-xs text-[#94a3b8]">Name</Label>
   418→            <Input
   419→              value={name}
   420→              onChange={(e) => handleNameChange(e.target.value)}
   421→              placeholder="e.g. Data Analyst Pro"
   422→              className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#64748b]"
   423→            />
   424→          </div>
   425→          <div className="space-y-2">
   426→            <Label className="text-xs text-[#94a3b8]">Slug</Label>
   427→            <Input
   428→              value={slug}
   429→              onChange={(e) => setSlug(e.target.value)}
   430→              placeholder="data-analyst-pro"
   431→              className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#64748b]"
   432→            />
   433→          </div>
   434→          <div className="space-y-2">
   435→            <Label className="text-xs text-[#94a3b8]">Description</Label>
   436→            <Textarea
   437→              value={description}
   438→              onChange={(e) => setDescription(e.target.value)}
   439→              placeholder="What does this agent do?"
   440→              rows={3}
   441→              className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#64748b] resize-none"
   442→            />
   443→          </div>
   444→          <div className="grid grid-cols-2 gap-4">
   445→            <div className="space-y-2">
   446→              <Label className="text-xs text-[#94a3b8]">Type</Label>
   447→              <Select value={type} onValueChange={setType}>
   448→                <SelectTrigger className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0]">
   449→                  <SelectValue />
   450→                </SelectTrigger>
   451→                <SelectContent className="glass-strong border-[rgba(99,102,241,0.18)]">
   452→                  {AGENT_TYPES.map((t) => (
   453→                    <SelectItem key={t} value={t} className="text-[#e2e8f0] focus:bg-[rgba(99,102,241,0.12)] focus:text-white">
   454→                      {t.charAt(0).toUpperCase() + t.slice(1)}
   455→                    </SelectItem>
   456→                  ))}
   457→                </SelectContent>
   458→              </Select>
   459→            </div>
   460→            <div className="space-y-2">
   461→              <Label className="text-xs text-[#94a3b8]">Status</Label>
   462→              <Select value={status} onValueChange={(v) => setStatus(v as AgentStatus)}>
   463→                <SelectTrigger className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0]">
   464→                  <SelectValue />
   465→                </SelectTrigger>
   466→                <SelectContent className="glass-strong border-[rgba(99,102,241,0.18)]">
   467→                  {AGENT_STATUSES.map((s) => (
   468→                    <SelectItem key={s} value={s} className="text-[#e2e8f0] focus:bg-[rgba(99,102,241,0.12)] focus:text-white">
   469→                      {s.charAt(0).toUpperCase() + s.slice(1)}
   470→                    </SelectItem>
   471→                  ))}
   472→                </SelectContent>
   473→              </Select>
   474→            </div>
   475→          </div>
   476→          <div className="space-y-2">
   477→            <Label className="text-xs text-[#94a3b8]">Capabilities</Label>
   478→            <Input
   479→              value={capabilities}
   480→              onChange={(e) => setCapabilities(e.target.value)}
   481→              placeholder="code-review, data-analysis, report-generation (comma-separated)"
   482→              className="bg-[rgba(99,102,241,0.06)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#64748b]"
   483→            />
   484→          </div>
   485→          <div className="flex justify-end gap-3 pt-2">
   486→            <Button
   487→              variant="outline"
   488→              className="border-[rgba(99,102,241,0.15)] text-[#94a3b8] hover:text-[#e2e8f0]"
   489→              onClick={() => onOpenChange(false)}
   490→            >
   491→              Cancel
   492→            </Button>
   493→            <Button
   494→              className="btn-gradient text-white"
   495→              onClick={handleSubmit}
   496→              disabled={!name.trim() || !slug.trim() || submitting}
   497→            >
   498→              {submitting ? 'Creating...' : 'Create Agent'}
   499→            </Button>
   500→          </div>
   501→        </div>
   502→      </DialogContent>
   503→    </Dialog>
   504→  )
   505→}

   507→function AgentDetailPanel({
   508→  agent,
   509→  open,
   510→  onOpenChange,
   511→  onDeleted,
   512→  orgId,
   513→}: {
   514→  agent: AgentDto | null
   515→  open: boolean
   516→  onOpenChange: (open: boolean) => void
   517→  onDeleted: () => void
   518→  orgId: string
   519→}) {
   520→  const capabilities = agent ? parseCapabilities(agent.capabilities) : []
   521→  const configuration = agent ? parseConfiguration(agent.configuration) : {}
   522→  const [deleting, setDeleting] = useState(false)
   523→
   524→  const handleDelete = async () => {
   525→    if (!agent) return
   526→    setDeleting(true)
   527→    try {
   528→      const res = await fetch(`/api/agents/${agent.id}?organizationId=${orgId}`, {
   529→        method: 'DELETE',
   530→      })
   531→      if (res.ok) {
   532→        toast({ title: 'Agent deleted', description: `${agent.name} has been removed.` })
   533→        onOpenChange(false)
   534→        onDeleted()
   535→      } else {
   536→        toast({ title: 'Delete failed', variant: 'destructive' })
   537→      }
   538→    } catch {
   539→      toast({ title: 'Error', description: 'Network error.', variant: 'destructive' })
   540→    } finally {
   541→      setDeleting(false)
   542→    }
   543→  }
   544→
   545→  if (!agent) return null
   546→
   547→  const configJson = JSON.stringify(configuration, null, 2)
   548→
   549→  return (
   550→    <Dialog open={open} onOpenChange={onOpenChange}>
   551→      <DialogContent className="glass-strong border-[rgba(99,102,241,0.18)] max-w-2xl max-h-[85vh]">
   552→        <DialogHeader>
   553→          <div className="flex items-center justify-between pr-6">
   554→            <DialogTitle className="text-[#e2e8f0] flex items-center gap-3">
   555→              <div className="w-9 h-9 rounded-lg bg-gradient-to-br from-[#6366f1] to-[#22d3ee] flex items-center justify-center">
   556→                <Bot className="w-4 h-4 text-white" />
   557→              </div>
   558→              {agent.name}
   559→            </DialogTitle>
   560→            <div className="flex items-center gap-1.5">
   561→              {getStatusBadge(agent.status)}
   562→            </div>
   563→          </div>
   564→        </DialogHeader>
   565→
   566→        <ScrollArea className="max-h-[60vh] pr-2">
   567→          <div className="space-y-5 pb-4">
   568→            {/* Basic Info */}

   686→export default function AgentsView() {
   687→  const activeOrg = useActiveOrg()
   688→  const isAuthenticated = useStore((s) => s.isAuthenticated)
   689→
   690→  const [agents, setAgents] = useState<AgentDto[]>([])
   691→  const [loading, setLoading] = useState(true)
   692→  const [error, setError] = useState<string | null>(null)
   693→  const [createOpen, setCreateOpen] = useState(false)
   694→  const [detailAgent, setDetailAgent] = useState<AgentDto | null>(null)
   695→  const [detailOpen, setDetailOpen] = useState(false)
   696→
   697→  const fetchAgents = useCallback(async () => {
   698→    if (!activeOrg?.id) return
   699→    setLoading(true)
   700→    setError(null)
   701→    try {
   702→      const res = await fetch(
   703→        `/api/agents?organizationId=${activeOrg.id}&limit=50`
   704→      )
   705→      if (res.ok) {
   706→        const json = await res.json()
   707→        setAgents(json.data ?? [])
   708→      } else {
   709→        setError('Failed to load agents')
   710→      }
   711→    } catch {
   712→      setError('Network error')
   713→    } finally {
   714→      setLoading(false)
   715→    }
   716→  }, [activeOrg?.id])
   717→
   718→  useEffect(() => {
   719→    if (activeOrg?.id) fetchAgents()
   720→  }, [activeOrg?.id, fetchAgents])
   721→
   722→  const handleSelectAgent = (agent: AgentDto) => {
   723→    setDetailAgent(agent)
   724→    setDetailOpen(true)
   725→  }
   726→
   727→  const handleRunAgent = (agent: AgentDto) => {
   728→    toast({
   729→      title: 'Agent run triggered',
   730→      description: `${agent.name} is now running.`,
   731→    })
   732→  }
   733→
   734→  const handleDelete = () => {
   735→    fetchAgents()
   736→  }
   737→
   738→  // Auth guard
   739→  if (!isAuthenticated) {
   740→    return (
   741→      <div className="space-y-6 animate-fade-in">

   743→        <motion.div
   744→          initial={{ opacity: 0, y: 20 }}
   745→          animate={{ opacity: 1, y: 0 }}
   746→          className="glass-card p-8 rounded-xl flex flex-col items-center justify-center min-h-[40vh]"
   747→        >
   748→          <AlertCircle className="w-10 h-10 text-[#f59e0b] mb-3 opacity-60" />
   749→          <p className="text-sm font-medium text-[#94a3b8]">
   750→            Sign in to manage your agents
   751→          </p>
   752→        </motion.div>
   753→      </div>
   754→    )
   755→  }
   756→
   757→  if (!activeOrg) {
   758→    return (
   759→      <div className="space-y-6 animate-fade-in">
   760→        <h1 className="text-2xl font-bold gradient-text">Agents</h1>
   761→        <motion.div
   762→          initial={{ opacity: 0, y: 20 }}
   763→          animate={{ opacity: 1, y: 0 }}
   764→          className="glass-card p-8 rounded-xl flex flex-col items-center justify-center min-h-[40vh]"
   765→        >
   766→          <Bot className="w-10 h-10 text-[#6366f1] mb-3 opacity-40" />
   767→          <p className="text-sm font-medium text-[#94a3b8]">
   768→            Select an organization to manage agents
   769→          </p>
   770→        </motion.div>
   771→      </div>
   772→    )
   773→  }
   774→
   775→  const totalAgents = agents.length
   776→  const activeAgents = agents.filter((a) => a.status === 'active').length
   777→

   779→    <>
   780→      <motion.div
   781→        className="space-y-6 animate-fade-in"
   782→        variants={containerVariants}
   783→        initial="hidden"
   784→        animate="visible"

   786→        {/* Header */}

   789→          className="flex items-center justify-between"

   791→          <h1 className="text-2xl font-bold gradient-text">Agents</h1>
   792→          <Button
   793→            className="btn-gradient text-white"
   794→            onClick={() => setCreateOpen(true)}
   795→          >
   796→            <Plus className="w-4 h-4 mr-1.5" />
   797→            Create Agent
   798→          </Button>
   799→        </motion.div>
   800→
   801→        {/* Stats Row */}
   802→        <StatsRow
   803→          total={totalAgents}
   804→          active={activeAgents}
   805→          runs={0}
   806→          loading={loading}
   807→        />
   808→
   809→        {/* Agents Grid */}
   810→        {loading ? (
   811→          <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   812→            {Array.from({ length: 6 }).map((_, i) => (
   813→              <Skeleton key={i} className="h-52 rounded-xl" />

   819→            className="glass-card p-8 rounded-xl text-center"

   821→            <AlertCircle className="w-10 h-10 text-[#ef4444] mx-auto mb-3 opacity-60" />

   824→            <Button
