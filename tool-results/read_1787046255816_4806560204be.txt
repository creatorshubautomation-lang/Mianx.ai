   187→function formatDate(dateStr: string | null): string {
   188→  if (!dateStr) return '—'
   189→  try {
   190→    return new Date(dateStr).toLocaleDateString('en-US', {
   191→      month: 'short', day: 'numeric', year: 'numeric',
   192→    })
   193→  } catch {
   194→    return '—'
   195→  }
   196→}
   197→
   198→function getInitials(name: string): string {
   199→  return name
   200→    .split(' ')
   201→    .map((w) => w[0])
   202→    .filter(Boolean)
   203→    .slice(0, 2)
   204→    .join('')
   205→    .toUpperCase()
   206→}

   334→function GeneralTab({
   335→  orgId,

   337→}: {
   338→  orgId: string

   340→}) {
   341→  const [form, setForm] = useState({
   342→    name: '', slug: '', timezone: 'UTC', locale: 'en', currency: 'USD',
   343→  })
   344→  const [saving, setSaving] = useState(false)
   345→  const [saved, setSaved] = useState(false)
   346→  const [error, setError] = useState<string | null>(null)
   347→  const [loading, setLoading] = useState(true)
   348→
   349→  useEffect(() => {
   350→    ;(async () => {
   351→      try {
   352→        const res = await fetch(`/api/organizations/${orgId}`)
   353→        const json: ApiResponseEnvelope<Record<string, string>> = await res.json()
   354→        if (json.error) throw new Error(json.error.message)
   355→        const d = json.data
   356→        setForm({
   357→          name: d.name ?? '',
   358→          slug: d.slug ?? '',
   359→          timezone: d.timezone ?? 'UTC',
   360→          locale: d.locale ?? 'en',
   361→          currency: d.currency ?? 'USD',
   362→        })
   363→      } catch (err) {
   364→        setError(err instanceof Error ? err.message : 'Failed to load')
   365→      } finally {
   366→        setLoading(false)
   367→      }
   368→    })()
   369→  }, [orgId])
   370→
   371→  const handleSave = async () => {
   372→    setSaving(true)
   373→    setError(null)
   374→    try {
   375→      const res = await fetch(`/api/organizations/${orgId}`, {
   376→        method: 'PUT',
   377→        headers: { 'Content-Type': 'application/json' },
   378→        body: JSON.stringify(form),
   379→      })
   380→      const json: ApiResponseEnvelope<Record<string, string>> = await res.json()
   381→      if (json.error) throw new Error(json.error.message)
   382→      setSaved(true)
   383→      setTimeout(() => setSaved(false), 2000)
   384→      // Refresh store
   385→      const listRes = await fetch('/api/organizations')
   386→      const listJson: ApiResponseEnvelope<import('@/lib/types').OrganizationDto[]> = await listRes.json()
   387→      if (!listJson.error && Array.isArray(listJson.data)) {
   388→        setOrganizations(listJson.data)
   389→      }
   390→    } catch (err) {
   391→      setError(err instanceof Error ? err.message : 'Save failed')
   392→    } finally {
   393→      setSaving(false)
   394→    }
   395→  }
   396→
   397→  if (loading) {
   398→    return (
   399→      <div className="glass-card p-6 rounded-xl space-y-4">
   400→        {Array.from({ length: 5 }).map((_, i) => (
   401→          <div key={i} className="space-y-2">
   402→            <Skeleton className="h-4 w-24 rounded" />
   403→            <Skeleton className="h-10 w-full rounded-lg" />
   404→          </div>
   405→        ))}
   406→      </div>
   407→    )
   408→  }
   409→
   410→  return (
   411→    <motion.div
   412→      initial={{ opacity: 0, y: 12 }}
   413→      animate={{ opacity: 1, y: 0 }}
   414→      className="glass-card p-6 rounded-xl space-y-5"

   416→      <h2 className="text-lg font-semibold text-[#e2e8f0]">General Settings</h2>

   418→
   419→      {/* Name */}
   420→      <div className="space-y-2">
   421→        <Label className="text-[#94a3b8] text-sm">Organization Name</Label>
   422→        <Input
   423→          value={form.name}
   424→          onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   425→          className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] max-w-md"
   426→        />
   427→      </div>
   428→
   429→      {/* Slug */}
   430→      <div className="space-y-2">
   431→        <Label className="text-[#94a3b8] text-sm">Slug</Label>

   433→          value={form.slug}

   437→      </div>
   438→

   443→          <SelectTrigger className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] max-w-md">
   444→            <SelectValue />

   494→      <div className="pt-2">
