   154→function relativeTime(dateStr: string | null): string {
   155→  if (!dateStr) return '—'
   156→  const diffMs = Date.now() - new Date(dateStr).getTime()
   157→  const seconds = Math.floor(diffMs / 1000)
   158→  if (seconds < 60) return 'just now'
   159→  const minutes = Math.floor(seconds / 60)
   160→  if (minutes < 60) return `${minutes}m ago`
   161→  const hours = Math.floor(minutes / 60)
   162→  if (hours < 24) return `${hours}h ago`
   163→  const days = Math.floor(hours / 24)
   164→  if (days < 30) return `${days}d ago`
   165→  return `${Math.floor(days / 30)}mo ago`
   166→}
   167→
   168→function formatDate(dateStr: string | null): string {
   169→  if (!dateStr) return '—'
   170→  return new Date(dateStr).toLocaleDateString('en-US', {
   171→    month: 'short', day: 'numeric', year: 'numeric',
   172→  })
   173→}
   174→
   175→function formatCost(amount: number): string {
   176→  if (amount === 0) return '$0.00'
   177→  return `$${amount.toFixed(2)}`
   178→}
   179→
   180→function formatVerificationType(type: string): string {
   181→  return type.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase())
   182→}
   183→
   184→function truncate(str: string, max: number): string {
   185→  if (!str) return ''
   186→  return str.length > max ? str.slice(0, max) + '...' : str
   187→}

   193→function EditMissionDialog({
   194→  mission,
   195→  open,
   196→  onOpenChange,
   197→  onSaved,
   198→}: {
   199→  mission: MissionDetailResponse | null
   200→  open: boolean
   201→  onOpenChange: (open: boolean) => void
   202→  onSaved: (updated: MissionDetailResponse) => void
   203→}) {
   204→  const [title, setTitle] = useState('')
   205→  const [goal, setGoal] = useState('')
   206→  const [budget, setBudget] = useState('')
   207→  const [deadline, setDeadline] = useState('')
   208→  const [submitting, setSubmitting] = useState(false)
   209→  const [error, setError] = useState('')
   210→  const addToast = useStore((s) => s.addToast)
   211→
   212→  useEffect(() => {
   213→    if (mission && open) {
   214→      setTitle(mission.title)
   215→      setGoal(mission.goal)
   216→      setBudget(mission.budget > 0 ? String(mission.budget) : '')
   217→      setDeadline(mission.deadline ? mission.deadline.slice(0, 10) : '')
   218→      setError('')
   219→    }
   220→  }, [mission, open])
   221→
   222→  const handleSave = async () => {
   223→    if (!mission) return
   224→    if (!title.trim() || !goal.trim()) {
   225→      setError('Title and goal are required')
   226→      return
   227→    }
   228→    setSubmitting(true)
   229→    setError('')
   230→    try {
   231→      const res = await fetch(`/api/missions/${mission.id}`, {
   232→        method: 'PUT',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({
   235→          title: title.trim(),
   236→          goal: goal.trim(),
   237→          budget: budget ? parseFloat(budget) : 0,
   238→          deadline: deadline || null,
   239→        }),
   240→      })
   241→      const json: ApiResponseEnvelope<MissionDetailResponse> = await res.json()
   242→      if (!res.ok || json.error) {
   243→        setError(json.error?.message ?? 'Failed to update')
   244→        return
   245→      }
   246→      addToast({ title: 'Mission updated', variant: 'success' })
   247→      onSaved(json.data)
   248→      onOpenChange(false)
   249→    } catch {
   250→      setError('Network error')
   251→    } finally {
   252→      setSubmitting(false)
   253→    }
   254→  }
   255→
   256→  return (
   257→    <Dialog open={open} onOpenChange={onOpenChange}>
   258→      <DialogContent className="glass-strong border-[rgba(99,102,241,0.18)] max-w-lg">
   259→        <DialogHeader>
   260→          <DialogTitle className="text-[#e2e8f0]">Edit Mission</DialogTitle>
   261→        </DialogHeader>
   262→        <div className="space-y-4 pt-2">
   263→          <div className="space-y-2">
   264→            <Label className="text-[#94a3b8] text-sm">Title</Label>
   265→            <Input value={title} onChange={(e) => setTitle(e.target.value)}
   266→              className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#475569]" />
   267→          </div>
   268→          <div className="space-y-2">
   269→            <Label className="text-[#94a3b8] text-sm">Goal</Label>
   270→            <Textarea value={goal} onChange={(e) => setGoal(e.target.value)} rows={3}
   271→              className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#475569] resize-none" />
   272→          </div>
   273→          <div className="grid grid-cols-2 gap-4">
   274→            <div className="space-y-2">
   275→              <Label className="text-[#94a3b8] text-sm">Budget ($)</Label>
   276→              <Input type="number" min="0" step="0.01" value={budget} onChange={(e) => setBudget(e.target.value)}
   277→                className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#475569]" />
   278→            </div>
   279→            <div className="space-y-2">
   280→              <Label className="text-[#94a3b8] text-sm">Deadline</Label>
   281→              <Input type="date" value={deadline} onChange={(e) => setDeadline(e.target.value)}
   282→                className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] placeholder:text-[#475569] [color-scheme:dark]" />
   283→            </div>
   284→          </div>
   285→          {error && (
   286→            <div className="flex items-center gap-2 text-xs text-[#ef4444] bg-[rgba(239,68,68,0.1)] border border-[rgba(239,68,68,0.2)] rounded-lg px-3 py-2">
   287→              <AlertCircle className="w-3.5 h-3.5 shrink-0" />{error}
   288→            </div>
   289→          )}
   290→          <div className="flex justify-end gap-3 pt-2">
   291→            <Button variant="ghost" onClick={() => onOpenChange(false)}
   292→              className="text-[#94a3b8] hover:text-[#e2e8f0]">Cancel</Button>
   293→            <Button onClick={handleSave} disabled={submitting} className="btn-gradient text-white border-0">
   294→              {submitting ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Pencil className="w-4 h-4 mr-2" />}
   295→              {submitting ? 'Saving...' : 'Save Changes'}
   296→            </Button>
   297→          </div>
   298→        </div>
   299→      </DialogContent>
   300→    </Dialog>
   301→  )
   302→}

   305→// Delete Confirm Dialog

   307→
   308→function DeleteConfirmDialog({
   309→  mission,
   310→  open,
   311→  onOpenChange,
   312→  onDeleted,
   313→}: {
   314→  mission: MissionDetailResponse | null
   315→  open: boolean
   316→  onOpenChange: (open: boolean) => void
   317→  onDeleted: () => void
   318→}) {
   319→  const [submitting, setSubmitting] = useState(false)
   320→  const [error, setError] = useState('')
   321→  const removeMission = useStore((s) => s.removeMission)
   322→  const addToast = useStore((s) => s.addToast)
   323→
   324→  const handleDelete = async () => {
   325→    if (!mission) return
   326→    setSubmitting(true)
   327→    setError('')
   328→    try {
   329→      const res = await fetch(`/api/missions/${mission.id}`, { method: 'DELETE' })
   330→      if (!res.ok && res.status !== 204) {
   331→        const json = await res.json().catch(() => null)
   332→        setError((json as { error?: { message?: string } })?.error?.message ?? 'Failed to delete')
   333→        return
   334→      }
   335→      removeMission(mission.id)
   336→      addToast({ title: 'Mission deleted', variant: 'success' })
   337→      onDeleted()
   338→      onOpenChange(false)
   339→    } catch {
   340→      setError('Network error')
   341→    } finally {
   342→      setSubmitting(false)
   343→    }
   344→  }
   345→
   346→  return (
   347→    <Dialog open={open} onOpenChange={onOpenChange}>
   348→      <DialogContent className="glass-strong border-[rgba(239,68,68,0.2)] max-w-sm">
   349→        <DialogHeader>
   350→          <DialogTitle className="text-[#e2e8f0]">Delete Mission?</DialogTitle>
   351→        </DialogHeader>
   352→        <p className="text-sm text-[#94a3b8]">
   353→          This will permanently delete &ldquo;{mission?.title}&rdquo; and all its tasks, outcomes, and verifications. This action cannot be undone.
   354→        </p>
   355→        {error && (
   356→          <div className="flex items-center gap-2 text-xs text-[#ef4444] bg-[rgba(239,68,68,0.1)] rounded-lg px-3 py-2">
   357→            <AlertCircle className="w-3.5 h-3.5 shrink-0" />{error}
   358→          </div>
   359→        )}
   360→        <div className="flex justify-end gap-3 pt-2">
   361→          <Button variant="ghost" onClick={() => onOpenChange(false)}
   362→            className="text-[#94a3b8] hover:text-[#e2e8f0]">Cancel</Button>
   363→          <Button onClick={handleDelete} disabled={submitting}
   364→            className="bg-[rgba(239,68,68,0.15)] text-[#ef4444] border border-[rgba(239,68,68,0.3)] hover:bg-[rgba(239,68,68,0.25)]">
   365→            {submitting ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Trash2 className="w-4 h-4 mr-2" />}
   366→            {submitting ? 'Deleting...' : 'Delete'}
   367→          </Button>
   368→        </div>
   369→      </DialogContent>
   370→    </Dialog>
   371→  )
   372→}
