   190→function formatDate(dateStr: string | null): string {
   191→  if (!dateStr) return '—'
   192→  try {
   193→    return new Date(dateStr).toLocaleDateString('en-US', {
   194→      month: 'short', day: 'numeric', year: 'numeric',
   195→    })
   196→  } catch {
   197→    return '—'
   198→  }
   199→}

   201→function getInitials(name: string): string {
   202→  return name
   203→    .split(' ')
   204→    .map((w) => w[0])
   205→    .filter(Boolean)
   206→    .slice(0, 2)
   207→    .join('')
   208→    .toUpperCase()
   209→}

   337→function GeneralTab({
   338→  orgId,
   339→  setOrganizations,
   340→}: {
   341→  orgId: string
   342→  setOrganizations: (orgs: import('@/lib/types').OrganizationDto[]) => void
   343→}) {
   344→  const [form, setForm] = useState({
   345→    name: '', slug: '', timezone: 'UTC', locale: 'en', currency: 'USD',
   346→  })
   347→  const [saving, setSaving] = useState(false)
   348→  const [saved, setSaved] = useState(false)
   349→  const [error, setError] = useState<string | null>(null)
   350→  const [loading, setLoading] = useState(true)
   351→
   352→  useEffect(() => {
   353→    ;(async () => {
   354→      try {
   355→        const res = await fetch(`/api/organizations/${orgId}`)
   356→        const json: ApiResponseEnvelope<Record<string, string>> = await res.json()
   357→        if (json.error) throw new Error(json.error.message)
   358→        const d = json.data
   359→        setForm({
   360→          name: d.name ?? '',
   361→          slug: d.slug ?? '',
   362→          timezone: d.timezone ?? 'UTC',
   363→          locale: d.locale ?? 'en',
   364→          currency: d.currency ?? 'USD',
   365→        })
   366→      } catch (err) {
   367→        setError(err instanceof Error ? err.message : 'Failed to load')
   368→      } finally {
   369→        setLoading(false)
   370→      }
   371→    })()
   372→  }, [orgId])
   373→
   374→  const handleSave = async () => {
   375→    setSaving(true)
   376→    setError(null)
   377→    try {
   378→      const res = await fetch(`/api/organizations/${orgId}`, {
   379→        method: 'PUT',
   380→        headers: { 'Content-Type': 'application/json' },
   381→        body: JSON.stringify(form),
   382→      })
   383→      const json: ApiResponseEnvelope<Record<string, string>> = await res.json()
   384→      if (json.error) throw new Error(json.error.message)
   385→      setSaved(true)
   386→      setTimeout(() => setSaved(false), 2000)
   387→      // Refresh store
   388→      const listRes = await fetch('/api/organizations')
   389→      const listJson: ApiResponseEnvelope<import('@/lib/types').OrganizationDto[]> = await listRes.json()
   390→      if (!listJson.error && Array.isArray(listJson.data)) {
   391→        setOrganizations(listJson.data)
   392→      }
   393→    } catch (err) {
   394→      setError(err instanceof Error ? err.message : 'Save failed')
   395→    } finally {
   396→      setSaving(false)
   397→    }
   398→  }
   399→
   400→  if (loading) {
   401→    return (
   402→      <div className="glass-card p-6 rounded-xl space-y-4">
   403→        {Array.from({ length: 5 }).map((_, i) => (
   404→          <div key={i} className="space-y-2">
   405→            <Skeleton className="h-4 w-24 rounded" />
   406→            <Skeleton className="h-10 w-full rounded-lg" />
   407→          </div>
   408→        ))}
   409→      </div>
   410→    )
   411→  }
   412→
   413→  return (
   414→    <motion.div
   415→      initial={{ opacity: 0, y: 12 }}
   416→      animate={{ opacity: 1, y: 0 }}
   417→      className="glass-card p-6 rounded-xl space-y-5"

   419→      <h2 className="text-lg font-semibold text-[#e2e8f0]">General Settings</h2>
   420→      <Separator className="bg-[rgba(99,102,241,0.12)]" />
   421→
   422→      {/* Name */}
   423→      <div className="space-y-2">
   424→        <Label className="text-[#94a3b8] text-sm">Organization Name</Label>
   425→        <Input
   426→          value={form.name}
   427→          onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   428→          className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] max-w-md"
   429→        />
   430→      </div>
   431→
   432→      {/* Slug */}
   433→      <div className="space-y-2">
   434→        <Label className="text-[#94a3b8] text-sm">Slug</Label>

   436→          value={form.slug}
   437→          onChange={(e) => setForm((f) => ({ ...f, slug: slugify(e.target.value) }))}
   438→          className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] max-w-md"
   439→        />
   440→      </div>
   441→
   442→      {/* Timezone */}
   443→      <div className="space-y-2">
   444→        <Label className="text-[#94a3b8] text-sm">Timezone</Label>
   445→        <Select value={form.timezone} onValueChange={(v) => setForm((f) => ({ ...f, timezone: v }))}>
   446→          <SelectTrigger className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.15)] text-[#e2e8f0] max-w-md">
   447→            <SelectValue />

   492→            <AlertCircle className="w-4 h-4 shrink-0" /> {error}

   495→      </AnimatePresence>
   496→
   497→      <div className="pt-2">
   498→        <Button
   499→          onClick={handleSave}
   500→          disabled={saving}

   509→      </div>
   510→    </motion.div>
   511→  )
   512→}
