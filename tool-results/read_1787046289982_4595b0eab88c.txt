   122→function formatRelativeTime(dateStr: string): string {
   123→  const date = new Date(dateStr)
   124→  const now = new Date()
   125→  const diffMs = now.getTime() - date.getTime()
   126→  const diffMins = Math.floor(diffMs / 60000)
   127→  const diffHours = Math.floor(diffMs / 3600000)
   128→  const diffDays = Math.floor(diffMs / 86400000)
   129→  if (diffMins < 1) return 'Just now'
   130→  if (diffMins < 60) return `${diffMins}m ago`
   131→  if (diffHours < 24) return `${diffHours}h ago`
   132→  if (diffDays < 7) return `${diffDays}d ago`
   133→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   134→}

   136→function formatCost(cents: number): string {
   137→  if (cents < 100) return `$${cents.toFixed(2)}`
   138→  return `$${(cents / 100).toFixed(2)}`
   139→}

   141→function formatTokens(n: number): string {
   142→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`
   143→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`
   144→  return String(n)
   145→}

   642→  const isAuthenticated = useStore((s) => s.isAuthenticated)
   643→  const [trustData, setTrustData] = useState<TrustData | null>(null)

   647→  const fetchTrustData = useCallback(async () => {
   648→    if (!activeOrg?.id) return
   649→    setLoading(true)
   650→    try {
   651→      const res = await fetch(`/api/trust?organizationId=${activeOrg.id}`)
   652→      if (res.ok) {
   653→        const json = await res.json()
   654→        setTrustData(json.data ?? null)
   655→      }
   656→    } catch {
   657→      // Silently handle
   658→    } finally {
   659→      setLoading(false)
   660→    }
   661→  }, [activeOrg?.id])
   662→
   663→  useEffect(() => {
   664→    if (!activeOrg?.id) return
   665→    fetchTrustData()
   666→  }, [activeOrg?.id, fetchTrustData])
   667→
   668→  if (!isAuthenticated) {
   669→    return (
   670→      <div className="space-y-6 animate-fade-in">
   671→        <h1 className="text-2xl font-bold gradient-text">Trust Center</h1>
   672→        <motion.div
   673→          initial={{ opacity: 0, y: 20 }}
   674→          animate={{ opacity: 1, y: 0 }}
   675→          className="glass-card p-8 rounded-xl flex flex-col items-center justify-center min-h-[40vh]"
   676→        >
   677→          <AlertTriangle className="w-10 h-10 text-[#f59e0b] mb-3 opacity-60" />
   678→          <p className="text-sm font-medium text-[#94a3b8]">Sign in to access the Trust Center</p>
   679→        </motion.div>
   680→      </div>
   681→    )
   682→  }
   683→
   684→  if (!activeOrg) {

   686→      <div className="space-y-6 animate-fade-in">
   687→        <h1 className="text-2xl font-bold gradient-text">Trust Center</h1>
   688→        <EmptyState />
   689→      </div>
   690→    )
   691→  }
   692→
   693→  const hasData = trustData && (
   694→    (trustData.recentExecutions?.length ?? 0) > 0 ||
   695→    (trustData.recentVerifications?.length ?? 0) > 0 ||
   696→    (trustData.recentApprovals?.length ?? 0) > 0 ||
   697→    (trustData.agentActionSummary?.totalRuns ?? 0) > 0
   698→  )
   699→
   700→  return (
   701→    <motion.div
   702→      className="space-y-6 animate-fade-in"
   703→      variants={containerVariants}
   704→      initial="hidden"
   705→      animate="visible"

   711→          See exactly what Mianx did, which agents acted, and what was verified.
   712→        </p>
   713→      </motion.div>
   714→
   715→      {/* Stats Row */}
   716→      <StatsRow data={trustData} loading={loading} />
   717→
   718→      {/* Tab Filters */}
   719→      <motion.div variants={itemVariants}>
   720→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   721→          <TabsList className="bg-[rgba(15,16,28,0.6)] border border-[rgba(99,102,241,0.1)]">
   722→            <TabsTrigger value="all" className="text-xs data-[state=active]:bg-[rgba(99,102,241,0.15)] data-[state=active]:text-[#e2e8f0]">
   723→              All
   724→            </TabsTrigger>

   740→              <>

   742→                <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
