    82→function formatDate(dateStr: string): string {
    83→  return new Date(dateStr).toLocaleDateString('en-US', {
    84→    month: 'short',
    85→    day: 'numeric',
    86→    year: 'numeric',
    87→  })
    88→}
    89→
    90→function formatCurrency(amount: number, currency: string = 'USD'): string {
    91→  return new Intl.NumberFormat('en-US', {
    92→    style: 'currency',
    93→    currency: currency.toUpperCase(),
    94→    minimumFractionDigits: 2,
    95→  }).format(amount / 100)
    96→}
    97→
    98→function getUsageColor(current: number, limit: number | null): string {
    99→  if (limit === null) return '#34d399'
   100→  const pct = (current / limit) * 100
   101→  if (pct > 90) return '#ef4444'
   102→  if (pct > 70) return '#f59e0b'
   103→  return '#34d399'
   104→}
   105→
   106→function getUsageProgress(current: number, limit: number | null): number {
   107→  if (limit === null) return 0
   108→  return Math.min(Math.round((current / limit) * 100), 100)
   109→}
   110→
   111→function getDaysRemaining(endsAt: string | null): number | null {
   112→  if (!endsAt) return null
   113→  const end = new Date(endsAt).getTime()
   114→  const now = Date.now()
   115→  const diff = end - now
   116→  if (diff <= 0) return 0
   117→  return Math.ceil(diff / 86400000)
   118→}
   119→
   120→function parseFeatures(featuresStr: string): string[] {
   121→  return parseJsonField<string[]>(featuresStr, [])
   122→}

   286→function UsageCard({
   287→  meter,
   288→  index,
   289→}: {
   290→  meter: { meterKey: string; meterName: string; unit: string; current: number; limit: number | null; period: string }
   291→  index: number
   292→}) {
   293→  const color = getUsageColor(meter.current, meter.limit)
   294→  const progress = getUsageProgress(meter.current, meter.limit)
   295→  const isUnlimited = meter.limit === null
   296→
   297→  const icons = [Zap, Activity, TrendingUp, CreditCard]
   298→  const Icon = icons[index % icons.length]
   299→
   300→  return (
   301→    <motion.div variants={itemVariants} className="glass-card p-5 rounded-xl">
   302→      <div className="flex items-start justify-between mb-3">
   303→        <div
   304→          className="w-9 h-9 rounded-lg flex items-center justify-center"
   305→          style={{ backgroundColor: `${color}18` }}
   306→        >
   307→          <Icon className="w-4.5 h-4.5" style={{ color }} />
   308→        </div>
   309→        {isUnlimited ? (
   310→          <Badge
   311→            variant="outline"
   312→            className="text-[9px] font-medium border-[rgba(99,102,241,0.15)] text-[#6366f1] bg-[rgba(99,102,241,0.06)] gap-1"
   313→          >
   314→            <Infinity className="w-3 h-3" />
   315→            Unlimited
   316→          </Badge>
   317→        ) : (
   318→          <span
   319→            className="text-xs font-semibold"
   320→            style={{ color }}
   321→          >
   322→            {progress}%
   323→          </span>
   324→        )}
   325→      </div>
   326→
   327→      <p className="text-xs text-[#94a3b8] mb-1">{meter.meterName}</p>
   328→      <div className="flex items-baseline gap-1.5 mb-3">
   329→        <span className="text-xl font-bold text-[#e2e8f0]">
   330→          {meter.current.toLocaleString()}
   331→        </span>
   332→        <span className="text-xs text-[#64748b]">
   333→          / {isUnlimited ? '∞' : meter.limit?.toLocaleString()} {meter.unit}
   334→        </span>
   335→      </div>
   336→
   337→      {!isUnlimited && (
   338→        <div className="space-y-1.5">
   339→          <Progress
   340→            value={progress}
   341→            className="h-1.5"
   342→          />
   343→          <p className="text-[10px] text-[#64748b]">
   344→            {meter.limit !== null && meter.limit - meter.current > 0
   345→              ? `${(meter.limit - meter.current).toLocaleString()} ${meter.unit} remaining this ${meter.period}`
   346→              : `Limit reached`}
   347→          </p>
   348→        </div>
   349→      )}
   350→    </motion.div>
   351→  )
   352→}

   557→export default function BillingView() {
   558→  const activeOrg = useActiveOrg()
   559→  const isAuthenticated = useStore((s) => s.isAuthenticated)
   560→
   561→  const [overview, setOverview] = useState<BillingOverview | null>(null)
   562→  const [invoices, setInvoices] = useState<InvoiceDto[]>([])
   563→  const [loading, setLoading] = useState(true)
   564→  const [error, setError] = useState<string | null>(null)
   565→
   566→  const fetchBilling = useCallback(async () => {
   567→    if (!activeOrg?.id) return
   568→    setLoading(true)
   569→    setError(null)
   570→    try {
   571→      const [overviewRes, invoicesRes] = await Promise.all([
   572→        fetch(`/api/billing/overview?organizationId=${activeOrg.id}`),
   573→        fetch(`/api/billing/invoices?organizationId=${activeOrg.id}&limit=20`),
   574→      ])
   575→
   576→      if (overviewRes.ok) {
   577→        const json = await overviewRes.json()
   578→        setOverview(json.data ?? null)
   579→      }
   580→
   581→      if (invoicesRes.ok) {
   582→        const json = await invoicesRes.json()
   583→        setInvoices(json.data ?? [])
   584→      }
   585→
   586→      if (!overviewRes.ok && !invoicesRes.ok) {
   587→        setError('Failed to load billing data')
   588→      }
   589→    } catch {
   590→      setError('Network error')
   591→    } finally {
   592→      setLoading(false)
   593→    }
   594→  }, [activeOrg?.id])
   595→
   596→  useEffect(() => {
   597→    if (activeOrg?.id) fetchBilling()
   598→  }, [activeOrg?.id, fetchBilling])
   599→
   600→  const showUpgrade =
   601→    !overview?.subscription ||
   602→    overview.subscription.status === 'trialing' ||
   603→    overview.subscription.status === 'past_due'
   604→
   605→  // Auth guard
   606→  if (!isAuthenticated) {
   607→    return (
   608→      <div className="space-y-6 animate-fade-in">
   609→        <h1 className="text-2xl font-bold gradient-text">Billing & Subscription</h1>
   610→        <motion.div
   611→          initial={{ opacity: 0, y: 20 }}
   612→          animate={{ opacity: 1, y: 0 }}
   613→          className="glass-card p-8 rounded-xl flex flex-col items-center justify-center min-h-[40vh]"
   614→        >
   615→          <AlertCircle className="w-10 h-10 text-[#f59e0b] mb-3 opacity-60" />
   616→          <p className="text-sm font-medium text-[#94a3b8]">
   617→            Sign in to view billing
   618→          </p>
   619→        </motion.div>
   620→      </div>
   621→    )
   622→  }
   623→
   624→  if (!activeOrg) {
   625→    return (
   626→      <div className="space-y-6 animate-fade-in">
   627→        <h1 className="text-2xl font-bold gradient-text">Billing & Subscription</h1>
   628→        <motion.div
   629→          initial={{ opacity: 0, y: 20 }}
   630→          animate={{ opacity: 1, y: 0 }}
   631→          className="glass-card p-8 rounded-xl flex flex-col items-center justify-center min-h-[40vh]"
   632→        >
   633→          <CreditCard className="w-10 h-10 text-[#6366f1] mb-3 opacity-40" />
   634→          <p className="text-sm font-medium text-[#94a3b8]">
   635→            Select an organization to view billing
   636→          </p>
   637→        </motion.div>
   638→      </div>
   639→    )
   640→  }
   641→
   642→  return (
   643→    <motion.div
   644→      className="space-y-6 animate-fade-in"
   645→      variants={containerVariants}
   646→      initial="hidden"
   647→      animate="visible"
   648→    >
   649→      {/* Header */}
   650→      <motion.div variants={itemVariants} className="flex items-center justify-between">
   651→        <h1 className="text-2xl font-bold gradient-text">Billing & Subscription</h1>
   652→        <Badge
   653→          variant="outline"
   654→          className="text-[10px] border-[rgba(99,102,241,0.2)] text-[#a78bfa] bg-[rgba(99,102,241,0.06)]"
   655→        >
   656→          <CreditCard className="w-3 h-3 mr-1" />
   657→          Manage
   658→        </Badge>
   659→      </motion.div>
   660→
   661→      {/* Error State */}
   662→      {error && !loading && (
   663→        <motion.div
   664→          variants={itemVariants}
   665→          className="glass-card p-8 rounded-xl text-center"
   666→        >
   667→          <AlertCircle className="w-10 h-10 text-[#ef4444] mx-auto mb-3 opacity-60" />
   668→          <p className="text-sm font-medium text-[#e2e8f0]">Failed to load billing data</p>
   669→          <p className="text-xs text-[#94a3b8] mt-1">{error}</p>
   670→          <Button
   671→            variant="outline"
   672→            className="mt-4 border-[rgba(99,102,241,0.15)] text-[#94a3b8] hover:text-[#e2e8f0]"

   683→      {/* Upgrade CTA */}

   685→        <motion.div
   686→          variants={itemVariants}
   687→          className="glass-strong rounded-xl p-6 flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4"
   688→        >
   689→          <div className="flex items-center gap-4">
   690→            <div className="w-12 h-12 rounded-xl bg-gradient-to-br from-[#6366f1] to-[#a78bfa] flex items-center justify-center flex-shrink-0">
