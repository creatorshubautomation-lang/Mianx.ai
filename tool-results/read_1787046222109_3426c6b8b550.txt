   176→function getMissionProgress(mission: MissionWithMeta): number {
   177→  const total = mission._count.tasks
   178→  if (total === 0) {
   179→    if (mission.status === 'completed') return 100
   180→    if (mission.status === 'executing') return 10
   181→    return 0
   182→  }
   183→  return Math.round((mission._completedTasks / total) * 100)
   184→}
   185→
   186→function formatRelativeTime(dateStr: string): string {
   187→  const date = new Date(dateStr)
   188→  const now = new Date()
   189→  const diffMs = now.getTime() - date.getTime()
   190→  const diffMins = Math.floor(diffMs / 60000)
   191→  const diffHours = Math.floor(diffMs / 3600000)
   192→  const diffDays = Math.floor(diffMs / 86400000)
   193→  if (diffMins < 1) return 'Just now'
   194→  if (diffMins < 60) return `${diffMins}m ago`
   195→  if (diffHours < 24) return `${diffHours}h ago`
   196→  if (diffDays < 7) return `${diffDays}d ago`
   197→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   198→}
   199→
   200→function formatCurrency(cents: number): string {
   201→  return `$${(cents / 100).toFixed(2)}`
   202→}

   233→function MissionInput({
   234→  onSubmit,
   235→  submitting,
   236→}: {
   237→  onSubmit: (data: { goal: string; budget?: string; deadline?: string; agentIds?: string[] }) => void
   238→  submitting: boolean
   239→}) {
   240→  const userMode = useUserMode()
   241→  const [goal, setGoal] = useState('')
   242→  const [budget, setBudget] = useState('')
   243→  const [deadline, setDeadline] = useState('')
   244→  const [selectedAgents, setSelectedAgents] = useState<string[]>([])
   245→  const [agents, setAgents] = useState<AgentOption[]>([])
   246→  const [agentsLoaded, setAgentsLoaded] = useState(false)
   247→  const activeOrg = useActiveOrg()
   248→
   249→  useEffect(() => {
   250→    if (!activeOrg?.id) return
   251→    fetch(`/api/agents?organizationId=${activeOrg.id}&limit=50`)
   252→      .then((r) => r.ok ? r.json() : null)
   253→      .then((json) => {
   254→        if (json?.data) {
   255→          setAgents(json.data.filter((a: AgentOption) => a.status === 'active').slice(0, 12))
   256→          setAgentsLoaded(true)
   257→        }
   258→      })
   259→      .catch(() => {})
   260→  }, [activeOrg?.id])
   261→
   262→  const handleSubmit = () => {
   263→    if (!goal.trim()) return
   264→    onSubmit({
   265→      goal: goal.trim(),
   266→      budget: budget || undefined,
   267→      deadline: deadline || undefined,
   268→      agentIds: selectedAgents.length > 0 ? selectedAgents : undefined,
   269→    })
   270→    setGoal('')
   271→    setBudget('')
   272→    setDeadline('')
   273→    setSelectedAgents([])
   274→  }
   275→
   276→  const toggleAgent = (id: string) => {
   277→    setSelectedAgents((prev) =>
   278→      prev.includes(id) ? prev.filter((a) => a !== id) : [...prev, id],
   279→    )
   280→  }
   281→
   282→  return (
   283→    <motion.div variants={itemVariants} className="glass-card p-6 sm:p-8 rounded-xl">
   284→      <div className="flex items-center gap-3 mb-5">
   285→        <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-[#6366f1] to-[#22d3ee] flex items-center justify-center">
   286→          <Rocket className="w-5 h-5 text-white" />
   287→        </div>
   288→        <div>
   289→          <h2 className="text-lg font-semibold text-[#e2e8f0]">Start a New Mission</h2>
   290→          <p className="text-xs text-[#94a3b8]">Describe what you want to accomplish</p>
   291→        </div>
   292→      </div>
   293→
   294→      <Textarea
   295→        placeholder="What do you want to accomplish?"
   296→        value={goal}
   297→        onChange={(e) => setGoal(e.target.value)}
   298→        className={cn(
   299→          'min-h-[100px] sm:min-h-[120px] resize-none text-[#e2e8f0] placeholder:text-[#64748b]',
   300→          'bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.12)] focus:border-[#6366f1]',
   301→          userMode === 'simple' ? 'text-base' : 'text-sm',
   302→        )}
   303→        onKeyDown={(e) => {
   304→          if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) handleSubmit()
   305→        }}
   306→      />
   307→
   308→      {/* Pro options */}
   309→      {userMode !== 'simple' && (
   310→        <motion.div
   311→          initial={{ opacity: 0, height: 0 }}
   312→          animate={{ opacity: 1, height: 'auto' }}
   313→          exit={{ opacity: 0, height: 0 }}
   314→          className="mt-4 space-y-4"
   315→        >
   316→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   317→            <div className="space-y-1.5">
   318→              <Label className="text-xs text-[#94a3b8] flex items-center gap-1.5">
   319→                <DollarSign className="w-3 h-3" /> Budget (USD)
   320→              </Label>
   321→              <Input
   322→                type="number"
   323→                placeholder="e.g. 10.00"
   324→                value={budget}
   325→                onChange={(e) => setBudget(e.target.value)}
   326→                className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.12)] text-[#e2e8f0] placeholder:text-[#64748b]"
   327→              />
   328→            </div>
   329→            <div className="space-y-1.5">
   330→              <Label className="text-xs text-[#94a3b8] flex items-center gap-1.5">
   331→                <CalendarDays className="w-3 h-3" /> Deadline
   332→              </Label>
   333→              <Input
   334→                type="date"
   335→                value={deadline}
   336→                onChange={(e) => setDeadline(e.target.value)}
   337→                className="bg-[rgba(15,16,28,0.6)] border-[rgba(99,102,241,0.12)] text-[#e2e8f0] placeholder:text-[#64748b]"
   338→              />
   339→            </div>
   340→          </div>
   341→
   342→          {/* Agent Selection */}
   343→          {agentsLoaded && agents.length > 0 && (
   344→            <div className="space-y-2">
   345→              <Label className="text-xs text-[#94a3b8] flex items-center gap-1.5">
   346→                <Bot className="w-3 h-3" /> Agent Selection
   347→              </Label>
   348→              <div className="flex flex-wrap gap-2">
   349→                {agents.map((agent) => {
   350→                  const isSelected = selectedAgents.includes(agent.id)
   351→                  return (
   352→                    <button
   353→                      key={agent.id}
   354→                      type="button"
   355→                      onClick={() => toggleAgent(agent.id)}
   356→                      className={cn(
   357→                        'px-3 py-1.5 rounded-lg text-xs font-medium transition-all border',
   358→                        isSelected
   359→                          ? 'bg-[rgba(99,102,241,0.2)] border-[#6366f1] text-[#e2e8f0]'
   360→                          : 'bg-[rgba(15,16,28,0.4)] border-[rgba(99,102,241,0.1)] text-[#94a3b8] hover:border-[rgba(99,102,241,0.25)] hover:text-[#e2e8f0]',
   361→                      )}
   362→                    >
   363→                      <Bot className="w-3 h-3 inline mr-1" />
   364→                      {agent.name}
   365→                    </button>
   366→                  )
   367→                })}
   368→              </div>
   369→            </div>
   370→          )}
   371→        </motion.div>
   372→      )}
   373→
   374→      <div className="flex items-center justify-between mt-5">
   375→        <p className="text-[10px] text-[#64748b]">
   376→          {userMode === 'simple' ? '' : 'Ctrl+Enter to submit'}
   377→        </p>
   378→        <Button
   379→          className={cn(
   380→            'btn-gradient text-white gap-2',
   381→            userMode === 'simple' ? 'px-8 py-5 text-base' : 'px-6 py-2.5 text-sm',
   382→          )}
   383→          disabled={!goal.trim() || submitting}
   384→          onClick={handleSubmit}
   385→        >
   386→          {submitting ? (
   387→            <Loader2 className="w-4 h-4 animate-spin" />
   388→          ) : (
   389→            <Send className="w-4 h-4" />
   390→          )}
   391→          {userMode === 'simple' ? 'Start Mission' : 'Create & Start'}
   392→        </Button>
   393→      </div>
   394→    </motion.div>
   395→  )
   396→}

   506→function ExpertTaskGraph({
   507→  mission,
   508→  loading,
   509→}: {
   510→  mission: MissionWithMeta
   511→  loading: boolean
   512→}) {
   513→  const [tasks, setTasks] = useState<MissionTaskItem[]>([])
   514→  const [tasksLoading, setTasksLoading] = useState(loading)
   515→  const [showJson, setShowJson] = useState(false)
   516→
   517→  useEffect(() => {
   518→    fetch(`/api/missions/${mission.id}/tasks?organizationId=${mission.id.includes('_') ? '' : ''}`)
   519→      .then((r) => r.ok ? r.json() : null)
   520→      .then((json) => {
   521→        if (json?.data) {
   522→          setTasks(json.data)
   523→        }
   524→        setTasksLoading(false)
   525→      })
   526→      .catch(() => setTasksLoading(false))
   527→  }, [mission.id])
   528→
   529→  // If tasks are passed in via mission prop, use those
   530→  const displayTasks = mission.tasks && mission.tasks.length > 0 ? mission.tasks : tasks
   531→
   532→  return (
   533→    <motion.div className="glass-card p-6 rounded-xl" variants={itemVariants}>
   534→      <div className="flex items-center justify-between mb-5">
   535→        <div className="flex items-center gap-2">

   538→          <span className="text-[10px] text-[#64748b]">{displayTasks.length} tasks</span>
   539→        </div>
   540→        <button
   541→          onClick={() => setShowJson(!showJson)}

   779→function ActiveMissions({
   780→  missions,
   781→  loading,
   782→}: {
   783→  missions: MissionWithMeta[]
   784→  loading: boolean
   785→}) {
   786→  const userMode = useUserMode()
   787→
   788→  if (loading) {
   789→    return (
   790→      <div className="space-y-4">
   791→        <Skeleton className="h-5 w-36" />
   792→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   793→          {Array.from({ length: 4 }).map((_, i) => (
   794→            <Skeleton key={i} className="h-44 rounded-xl" />
   795→          ))}
   796→        </div>
   797→      </div>
   798→    )
   799→  }
   800→
   801→  const activeMissions = missions.filter(
   802→    (m) => m.status !== 'cancelled' && m.status !== 'draft',
   803→  )
   804→
   805→  if (activeMissions.length === 0) {
   806→    return null
   807→  }
   808→
   809→  return (
   810→    <div className="space-y-4">
   811→      <div className="flex items-center gap-2">
   812→        <Rocket className="w-4 h-4 text-[#6366f1]" />
   813→        <h3 className="text-base font-semibold text-[#e2e8f0]">Active Missions</h3>
   814→        <Badge
   815→          variant="secondary"
   816→          className="text-[10px] font-medium border-0"
   817→          style={{ color: '#6366f1', backgroundColor: 'rgba(99,102,241,0.12)' }}
   818→        >
   819→          {activeMissions.length}
   820→        </Badge>
   821→      </div>
   822→
   823→      {userMode === 'simple' ? (
   824→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   825→          {activeMissions.map((m) => (
   826→            <SimpleMissionCard key={m.id} mission={m} />
   827→          ))}
   828→        </div>
   829→      ) : userMode === 'pro' ? (
   830→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   831→          {activeMissions.map((m) => (
   832→            <ProMissionCard key={m.id} mission={m} />
   833→          ))}
   834→        </div>
   835→      ) : (
   836→        <div className="space-y-4">
   837→          {activeMissions.map((m) => (
   838→            <ExpertMissionCard key={m.id} mission={m} />
   839→          ))}
   840→        </div>
   841→      )}
   842→    </div>
   843→  )
   844→}

   870→      setLoading(false)

   875→    if (!activeOrg?.id) return

   902→        const json = await res.json()
   903→        const newMission = json.data
   904→        if (newMission?.id) {
   905→          navigate('mission-detail', { missionId: newMission.id })
   906→        }
   907→        fetchMissions()
   908→      }
   909→    } catch {
   910→      // silently handle
   911→    } finally {
   912→      setSubmitting(false)
   913→    }
   914→  }
   915→
   916→  if (!isAuthenticated) {
   917→    return (
   918→      <div className="space-y-6 animate-fade-in">
   919→        <h1 className="text-2xl font-bold gradient-text">Command Center</h1>
   920→        <motion.div
   921→          initial={{ opacity: 0, y: 20 }}

   932→  if (!activeOrg) {

   934→      <div className="space-y-6 animate-fade-in">
   935→        <h1 className="text-2xl font-bold gradient-text">Command Center</h1>
